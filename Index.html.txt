        function getRowValue(row, colName) {
            if (!row || !colName) return '';
            if (row[colName] !== undefined) return row[row[colName] !== undefined ? colName : ''];
            
            // Loose matching for whitespace or casing differences
            const cleanTarget = colName.toLowerCase().replace(/\s+/g, '');
            for (let key in row) {
                if (key.toLowerCase().replace(/\s+/g, '') === cleanTarget) {
                    return row[key];
                }
            }
            return '';
        }

        function parseDateDetails(dateStr) {
            if (!dateStr) return { year: null, monthIndex: null, monthName: null };
            const str = String(dateStr).trim();
            const lowerStr = str.toLowerCase();

            const monthNames = [
                'enero', 'febrero', 'marzo', 'abril', 'mayo', 'junio',
                'julio', 'agosto', 'septiembre', 'octubre', 'noviembre', 'diciembre'
            ];
            const monthAbbrs = [
                'ene', 'feb', 'mar', 'abr', 'may', 'jun',
                'jul', 'ago', 'sep', 'oct', 'nov', 'dic'
            ];

            // Match month names
            for (let i = 0; i < 12; i++) {
                if (lowerStr.includes(monthNames[i]) || lowerStr.startsWith(monthAbbrs[i])) {
                    const yearMatch = str.match(/20\d\d/);
                    const year = yearMatch ? parseInt(yearMatch[0], 10) : null;
                    return { year: year, monthIndex: i, monthName: monthNames[i].charAt(0).toUpperCase() + monthNames[i].slice(1) };
                }
            }

            // Numerical date format: DD/MM/YYYY or YYYY-MM-DD
            let day, month, year;
            if (str.includes('-')) {
                const parts = str.split('-');
                if (parts[0].length === 4) { // YYYY-MM-DD
                    year = parseInt(parts[0], 10);
                    month = parseInt(parts[1], 10) - 1;
                } else { // DD-MM-YYYY
                    month = parseInt(parts[1], 10) - 1;
                    year = parseInt(parts[2], 10);
                }
            } else if (str.includes('/')) {
                const parts = str.split('/');
                if (parts[2] && parts[2].length >= 2) {
                    month = parseInt(parts[1], 10) - 1;
                    year = parseInt(parts[2], 10);
                    if (year < 100) year += 2000;
                }
            }

            if (month >= 0 && month < 12) {
                const spanishMonths = ['Enero', 'Febrero', 'Marzo', 'Abril', 'Mayo', 'Junio', 'Julio', 'Agosto', 'Septiembre', 'Octubre', 'Noviembre', 'Diciembre'];
                return { year: year || null, monthIndex: month, monthName: spanishMonths[month] };
            }

            // Standard JS Date
            const d = new Date(str);
            if (!isNaN(d.getTime())) {
                const spanishMonths = ['Enero', 'Febrero', 'Marzo', 'Abril', 'Mayo', 'Junio', 'Julio', 'Agosto', 'Septiembre', 'Octubre', 'Noviembre', 'Diciembre'];
                return { year: d.getFullYear(), monthIndex: d.getMonth(), monthName: spanishMonths[d.getMonth()] };
            }

            return { year: null, monthIndex: null, monthName: null };
        }

                        return false;
                    }
                }
                // Search Text Filter
                if (searchText !== '') {
                    const rowContent = Object.values(row).join(' ').toLowerCase();
                    if (!rowContent.includes(searchText)) return false;
                }
                return true;
            });

            renderKPIs();
            renderCharts();
            renderTable();

            if (currentTab === 'yoy') {
                updateYoYView();
            }
        }

        function renderKPIs() {
            let totalSales = 0;
            let totalTx = filteredData2026.length;
            let totalQty = 0;
            const storeTotals = {};
            const productTotals = {};

            filteredData2026.forEach(row => {
                const salesVal = parseNum(getRowValue(row, columnMapping.sales));
                totalSales += salesVal;

                // Sum directly from "cantidad" column
                const qVal = parseNum(getRowValue(row, columnMapping.qty));
                totalQty += qVal;

                // Store aggregation
                const store = getRowValue(row, columnMapping.store) || 'Sin Especificar';
                storeTotals[store] = (storeTotals[store] || 0) + salesVal;

                // Product aggregation
                const prod = getRowValue(row, columnMapping.product) || getRowValue(row, columnMapping.productType) || 'Sin Nombre';
                if (!productTotals[prod]) {
                    productTotals[prod] = { amount: 0, count: 0 };
                }
                productTotals[prod].amount += salesVal;
                productTotals[prod].count += 1;
            });

            // Top Store
            let topStore = '-';
            let topStoreAmt = 0;
            for (let s in storeTotals) {
                if (storeTotals[s] > topStoreAmt) {
                    topStoreAmt = storeTotals[s];
                    topStore = s;
                }
            }

            // Top Product
            let topProd = '-';
            let topProdAmt = 0;
            let topProdCount = 0;
            for (let p in productTotals) {
                if (productTotals[p].amount > topProdAmt) {
                    topProdAmt = productTotals[p].amount;
                    topProdCount = productTotals[p].count;
                    topProd = p;
                }
            }

            // Render KPI elements safely
            if (document.getElementById('kpiTotalSales')) document.getElementById('kpiTotalSales').innerText = formatCurrencyNoDecimals(totalSales);
            if (document.getElementById('kpiTotalQty')) document.getElementById('kpiTotalQty').innerText = Math.round(totalQty).toLocaleString('en-US');
            if (document.getElementById('kpiAvgQty')) document.getElementById('kpiAvgQty').innerText = 'Prom. por Venta: ' + (totalTx > 0 ? (totalQty / totalTx).toFixed(1) : 0) + ' unid.';
            
            if (document.getElementById('kpiTopStore')) document.getElementById('kpiTopStore').innerText = topStore;
            if (document.getElementById('kpiTopStoreAmount')) document.getElementById('kpiTopStoreAmount').innerText = formatCurrencyNoDecimals(topStoreAmt) + ' en ventas';
            
            const kpiTopProdElem = document.getElementById('kpiTopProduct');
            if (kpiTopProdElem) {
                kpiTopProdElem.innerText = topProd;
                kpiTopProdElem.title = topProd;
            }
            if (document.getElementById('kpiTopProductAmount')) {
                document.getElementById('kpiTopProductAmount').innerText = `${topProdCount} unid. (${formatCurrencyNoDecimals(topProdAmt)})`;
            }
        }

        function updateYoYView() {
            const isDark = document.body.classList.contains('dark-mode');
            const textColor = isDark ? '#f8fafc' : '#475569';
            const gridColor = isDark ? 'rgba(255, 255, 255, 0.1)' : 'rgba(0, 0, 0, 0.05)';

            // Read active filter constraints
            const selectedTienda = document.getElementById('filterTienda')?.value || 'ALL';
            const selectedAmbiente = document.getElementById('filterAmbiente')?.value || 'ALL';
            const selectedTipoProducto = document.getElementById('filterTipoProducto')?.value || 'ALL';

            // Filter 2026 dataset
            const filtered2026 = rawData2026.filter(r => {
                if (selectedTienda !== 'ALL' && getRowValue(r, columnMapping.store) !== selectedTienda) return false;
                if (selectedAmbiente !== 'ALL' && getRowValue(r, columnMapping.env) !== selectedAmbiente) return false;
                if (selectedTipoProducto !== 'ALL' && getRowValue(r, columnMapping.productType) !== selectedTipoProducto) return false;
                return true;
            });

            // Filter 2025 dataset by active filter selections
            const filtered2025 = rawData2025.filter(r => {
                if (selectedTienda !== 'ALL' && getRowValue(r, columnMapping.store) !== selectedTienda) return false;
                if (selectedAmbiente !== 'ALL' && getRowValue(r, columnMapping.env) !== selectedAmbiente) return false;
                if (selectedTipoProducto !== 'ALL' && getRowValue(r, columnMapping.productType) !== selectedTipoProducto) return false;
                return true;
            });

            // Find last active month in 2026 for aligned period comparison
            let maxMonth2026 = 0;
            const monthly2026 = new Array(12).fill(0);
            const tx2026 = new Array(12).fill(0);
            const qty2026 = new Array(12).fill(0);

            filtered2026.forEach(r => {
                let d = parseDateDetails(getRowValue(r, columnMapping.date));
                if (d.monthIndex !== null && d.monthIndex >= 0 && d.monthIndex < 12) {
                    monthly2026[d.monthIndex] += parseNum(getRowValue(r, columnMapping.sales));
                    tx2026[d.monthIndex] += 1;
                    
                    // Sum directly from "cantidad" column
                    const qVal = parseNum(getRowValue(r, columnMapping.qty));
                    qty2026[d.monthIndex] += qVal;
                    if (d.monthIndex > maxMonth2026) maxMonth2026 = d.monthIndex;
                }
            });

            // Filter 2025 data strictly up to maxMonth2026
            const monthly2025 = new Array(12).fill(0);
            const tx2025 = new Array(12).fill(0);
            const qty2025 = new Array(12).fill(0);

            const filtered2025Aligned = filtered2025.filter(r => {
                let d = parseDateDetails(getRowValue(r, columnMapping.date));
                return d.monthIndex !== null && d.monthIndex >= 0 && d.monthIndex <= maxMonth2026;
            });

            filtered2025Aligned.forEach(r => {
                let d = parseDateDetails(getRowValue(r, columnMapping.date));
                if (d.monthIndex !== null && d.monthIndex >= 0 && d.monthIndex < 12) {
                    monthly2025[d.monthIndex] += parseNum(getRowValue(r, columnMapping.sales));
                    tx2025[d.monthIndex] += 1;
                    
                    // Sum directly from "cantidad" column
                    const qVal = parseNum(getRowValue(r, columnMapping.qty));
                    qty2025[d.monthIndex] += qVal;
                }
            });

            // Sum totals up to maxMonth2026 for aligned period comparison
            let sum2026 = 0;
            let sum2025 = 0;
            let count2026 = 0;
            let count2025 = 0;
            let totalQty2026 = 0;
            let totalQty2025 = 0;

            for (let i = 0; i <= maxMonth2026; i++) {
                sum2026 += monthly2026[i];
                sum2025 += monthly2025[i];
                count2026 += tx2026[i];
                count2025 += tx2025[i];
                totalQty2026 += qty2026[i];
                totalQty2025 += qty2025[i];
            }

            const diffAmt = sum2026 - sum2025;
            const pctGrowth = sum2025 > 0 ? ((diffAmt / sum2025) * 100) : 0;

            const qtyDiff = totalQty2026 - totalQty2025;
            const qtyPct = totalQty2025 > 0 ? ((qtyDiff / totalQty2025) * 100) : 0;

            // Update Card 1: Ventas YoY
            if (document.getElementById('yoy2026Total')) document.getElementById('yoy2026Total').innerText = formatCurrencyNoDecimals(sum2026);
            if (document.getElementById('yoy2025Total')) document.getElementById('yoy2025Total').innerText = formatCurrencyNoDecimals(sum2025);
            if (document.getElementById('yoyDiffAmount')) document.getElementById('yoyDiffAmount').innerText = (diffAmt >= 0 ? '+' : '') + formatCurrencyNoDecimals(diffAmt);
            
            const badgeSales = document.getElementById('yoySalesBadge');
            if (badgeSales) {
                if (pctGrowth > 0) {
                    badgeSales.className = 'inline-flex items-center text-xs font-semibold px-2 py-0.5 rounded-full bg-emerald-50 text-emerald-700 border border-emerald-200';
                    badgeSales.innerHTML = `<i class="fa-solid fa-arrow-trend-up mr-1"></i>+${pctGrowth.toFixed(1)}%`;
                } else if (pctGrowth < 0) {
                    badgeSales.className = 'inline-flex items-center text-xs font-semibold px-2 py-0.5 rounded-full bg-rose-50 text-rose-700 border border-rose-200';
                    badgeSales.innerHTML = `<i class="fa-solid fa-arrow-trend-down mr-1"></i>${pctGrowth.toFixed(1)}%`;
                } else {
                    badgeSales.className = 'inline-flex items-center text-xs font-semibold px-2 py-0.5 rounded-full bg-slate-100 text-slate-600';
                    badgeSales.innerHTML = `<i class="fa-solid fa-minus mr-1"></i>0.0%`;
                }
            }

            // Update Card 2: Cantidad YoY
            if (document.getElementById('yoyQty2026')) document.getElementById('yoyQty2026').innerText = Math.round(totalQty2026).toLocaleString('en-US');
            if (document.getElementById('yoyQty2025')) document.getElementById('yoyQty2025').innerText = Math.round(totalQty2025).toLocaleString('en-US');
            if (document.getElementById('yoyQtyDiff')) document.getElementById('yoyQtyDiff').innerText = (qtyDiff >= 0 ? '+' : '') + Math.round(qtyDiff).toLocaleString('en-US');

            const badgeQty = document.getElementById('yoyQtyBadge');
            if (badgeQty) {
                if (qtyPct > 0) {
                    badgeQty.className = 'inline-flex items-center text-xs font-semibold px-2 py-0.5 rounded-full bg-emerald-50 text-emerald-700 border border-emerald-200';
                    badgeQty.innerHTML = `<i class="fa-solid fa-arrow-trend-up mr-1"></i>+${qtyPct.toFixed(1)}%`;
                } else if (qtyPct < 0) {
                    badgeQty.className = 'inline-flex items-center text-xs font-semibold px-2 py-0.5 rounded-full bg-rose-50 text-rose-700 border border-rose-200';
                    badgeQty.innerHTML = `<i class="fa-solid fa-arrow-trend-down mr-1"></i>${qtyPct.toFixed(1)}%`;
                } else {
                    badgeQty.className = 'inline-flex items-center text-xs font-semibold px-2 py-0.5 rounded-full bg-slate-100 text-slate-600';
                    badgeQty.innerHTML = `<i class="fa-solid fa-minus mr-1"></i>0.0%`;
                }
            }

            // Update Card 3: Producto Estrella YoY
            const yoyProdMap2026 = {};
            const yoyProdMap2025 = {};

            filtered2026.forEach(r => {
                const prod = getRowValue(r, columnMapping.product) || getRowValue(r, columnMapping.productType) || 'Sin Especificar';
                const sVal = parseNum(getRowValue(r, columnMapping.sales));
                const qVal = parseNum(getRowValue(r, columnMapping.qty));
                if (!yoyProdMap2026[prod]) yoyProdMap2026[prod] = { sales: 0, qty: 0 };
                yoyProdMap2026[prod].sales += sVal;
                yoyProdMap2026[prod].qty += qVal;
            });

            filtered2025Aligned.forEach(r => {
                const prod = getRowValue(r, columnMapping.product) || getRowValue(r, columnMapping.productType) || 'Sin Especificar';
                const sVal = parseNum(getRowValue(r, columnMapping.sales));
                const qVal = parseNum(getRowValue(r, columnMapping.qty));
                if (!yoyProdMap2025[prod]) yoyProdMap2025[prod] = { sales: 0, qty: 0 };
                yoyProdMap2025[prod].sales += sVal;
                yoyProdMap2025[prod].qty += qVal;
            });

            let yoyTopProdName = '-';
            let yoyTopProdSales2026 = 0;
            let yoyTopProdQty2026 = 0;

            for (let p in yoyProdMap2026) {
                if (yoyProdMap2026[p].sales > yoyTopProdSales2026) {
                    yoyTopProdSales2026 = yoyProdMap2026[p].sales;
                    yoyTopProdQty2026 = yoyProdMap2026[p].qty;
                    yoyTopProdName = p;
                }
            }

            let yoyTopProdSales2025 = 0;
            let yoyTopProdQty2025 = 0;
            if (yoyTopProdName !== '-' && yoyProdMap2025[yoyTopProdName]) {
                yoyTopProdSales2025 = yoyProdMap2025[yoyTopProdName].sales;
                yoyTopProdQty2025 = yoyProdMap2025[yoyTopProdName].qty;
            }

            const yoyTopProdDiff = yoyTopProdSales2026 - yoyTopProdSales2025;
            const yoyTopProdPct = yoyTopProdSales2025 > 0 ? ((yoyTopProdDiff / yoyTopProdSales2025) * 100) : 0;

            if (document.getElementById('yoyTopProductName')) {
                const elem = document.getElementById('yoyTopProductName');
                elem.innerText = yoyTopProdName;
                elem.title = yoyTopProdName;
            }
            if (document.getElementById('yoyTopProduct2026')) {
                document.getElementById('yoyTopProduct2026').innerHTML = `2026: <span class="font-semibold text-slate-700">${formatCurrencyNoDecimals(yoyTopProdSales2026)} (${Math.round(yoyTopProdQty2026).toLocaleString('en-US')} unid.)</span>`;
            }
            if (document.getElementById('yoyTopProduct2025')) {
                document.getElementById('yoyTopProduct2025').innerText = formatCurrencyNoDecimals(yoyTopProdSales2025);
            }

            const badgeTopProd = document.getElementById('yoyTopProductBadge');
            if (badgeTopProd) {
                if (yoyTopProdSales2025 === 0 && yoyTopProdSales2026 > 0) {
                    badgeTopProd.className = 'inline-flex items-center text-xs font-semibold px-2 py-0.5 rounded-full bg-emerald-50 text-emerald-700 border border-emerald-200';
                    badgeTopProd.innerHTML = `<i class="fa-solid fa-star mr-1"></i>Nuevo`;
                } else if (yoyTopProdPct > 0) {
                    badgeTopProd.className = 'inline-flex items-center text-xs font-semibold px-2 py-0.5 rounded-full bg-emerald-50 text-emerald-700 border border-emerald-200';
                    badgeTopProd.innerHTML = `<i class="fa-solid fa-arrow-trend-up mr-1"></i>+${yoyTopProdPct.toFixed(1)}%`;
                } else if (yoyTopProdPct < 0) {
                    badgeTopProd.className = 'inline-flex items-center text-xs font-semibold px-2 py-0.5 rounded-full bg-rose-50 text-rose-700 border border-rose-200';

            // Render YoY Monthly Chart only for elapsed months (0 to maxMonth2026)
            const monthNames = ['Ene', 'Feb', 'Mar', 'Abr', 'May', 'Jun', 'Jul', 'Ago', 'Sep', 'Oct', 'Nov', 'Dic'];
            const displayMonthNames = monthNames.slice(0, maxMonth2026 + 1);
            const displayMonthly2026 = monthly2026.slice(0, maxMonth2026 + 1);
            const displayMonthly2025 = monthly2025.slice(0, maxMonth2026 + 1);

            if (chartYoYMonthlyInst) chartYoYMonthlyInst.destroy();

                                label: '2026',

                                borderRadius: 6
                            }
                        ]
                    },
                    options: {
                        responsive: true,
                        maintainAspectRatio: false,
                        layout: {
                            padding: { top: 20 }

                                formatter: (v) => {

            // Render YoY Store Chart using strictly aligned 2025 data (up to maxMonth2026)
            const storeMap2026 = {};

                storeMap2026[s] = (storeMap2026[s] || 0) + parseNum(getRowValue(r, columnMapping.sales));

            filtered2025Aligned.forEach(r => {
                const s = getRowValue(r, columnMapping.store) || 'Sin Especificar';
                storeMap2025[s] = (storeMap2025[s] || 0) + parseNum(getRowValue(r, columnMapping.sales));

            const allStores = Array.from(new Set([...Object.keys(storeMap2026), ...Object.keys(storeMap2025)])).sort();
            const storeValues2026 = allStores.map(s => storeMap2026[s] || 0);

                                label: '2026',

            // Render YoY Product Breakdown Table
            const yoyTbody = document.getElementById('yoyTableBody');
            if (yoyTbody) {

                const productMap2026 = {};
                const productMap2025 = {};
                const productTypeMap = {};

                filtered2026.forEach(r => {
                    const prod = getRowValue(r, columnMapping.product) || getRowValue(r, columnMapping.productType) || 'Sin Especificar';
                    const pType = getRowValue(r, columnMapping.productType) || '-';
                    const sVal = parseNum(getRowValue(r, columnMapping.sales));
                    const qVal = parseNum(getRowValue(r, columnMapping.qty));

                    if (!productMap2026[prod]) productMap2026[prod] = { sales: 0, qty: 0 };
                    productMap2026[prod].sales += sVal;
                    productMap2026[prod].qty += (qVal > 0 ? qVal : 1);
                    productTypeMap[prod] = pType;
                });

                filtered2025Aligned.forEach(r => {
                    const prod = getRowValue(r, columnMapping.product) || getRowValue(r, columnMapping.productType) || 'Sin Especificar';
                    const pType = getRowValue(r, columnMapping.productType) || '-';
                    const sVal = parseNum(getRowValue(r, columnMapping.sales));
                    const qVal = parseNum(getRowValue(r, columnMapping.qty));

                    if (!productMap2025[prod]) productMap2025[prod] = { sales: 0, qty: 0 };
                    productMap2025[prod].sales += sVal;
                    productMap2025[prod].qty += (qVal > 0 ? qVal : 1);
                    if (!productTypeMap[prod]) productTypeMap[prod] = pType;
                });

                const allProducts = Array.from(new Set([...Object.keys(productMap2026), ...Object.keys(productMap2025)]));

                // Sort by 2026 sales descending
                allProducts.sort((a, b) => {
                    const salesA = productMap2026[a] ? productMap2026[a].sales : 0;
                    const salesB = productMap2026[b] ? productMap2026[b].sales : 0;
                    return salesB - salesA;
                });

                if (allProducts.length === 0) {
                    yoyTbody.innerHTML = `<tr><td colspan="7" class="p-6 text-center text-slate-400">No hay datos de productos para mostrar</td></tr>`;
                } else {
                    allProducts.forEach(prodName => {
                        const val26 = productMap2026[prodName] ? productMap2026[prodName].sales : 0;
                        const qty26 = productMap2026[prodName] ? productMap2026[prodName].qty : 0;
                        const val25 = productMap2025[prodName] ? productMap2025[prodName].sales : 0;
                        const qty25 = productMap2025[prodName] ? productMap2025[prodName].qty : 0;

                        const diff = val26 - val25;
                        const pct = val25 > 0 ? ((diff / val25) * 100) : 0;
                        const prodType = productTypeMap[prodName] || '-';

                        const tr = document.createElement('tr');
                        tr.className = 'hover:bg-slate-50 transition';

                        const pctColor = diff > 0 ? 'text-emerald-500 font-bold' : (diff < 0 ? 'text-rose-500 font-bold' : 'text-slate-400');

                        tr.innerHTML = `
                            <td class="p-3 sm:p-4 font-semibold text-slate-900">${prodName}</td>
                            <td class="p-3 sm:p-4"><span class="px-2 py-0.5 bg-slate-100 text-slate-700 rounded-md text-xs">${prodType}</span></td>
                            <td class="p-3 sm:p-4 text-right font-medium">${formatCurrency(val26)}</td>
                            <td class="p-3 sm:p-4 text-right text-slate-500">${formatCurrency(val25)}</td>
                            <td class="p-3 sm:p-4 text-right ${diff >= 0 ? 'text-emerald-500 font-semibold' : 'text-rose-500 font-semibold'}">${(diff >= 0 ? '+' : '') + formatCurrency(diff)}</td>
                            <td class="p-3 sm:p-4 text-right ${pctColor}">${val25 > 0 ? (pct >= 0 ? '+' : '') + pct.toFixed(1) + '%' : (val26 > 0 ? 'Nuevo' : '-')}</td>
                            <td class="p-3 sm:p-4 text-right text-slate-500">${qty26.toLocaleString()} vs ${qty25.toLocaleString()}</td>
                        `;
                        yoyTbody.appendChild(tr);
                    });
                }
            }
        }

            currentTab = tab;
            const viewMain = document.getElementById('viewMain');
            const viewYoY = document.getElementById('viewYoY');
            const tabMainBtn = document.getElementById('tabMainBtn');
            const tabYoYBtn = document.getElementById('tabYoYBtn');

                viewMain.classList.remove('hidden');
                viewYoY.classList.add('hidden');
                tabMainBtn.className = 'px-3 py-1.5 sm:px-4 sm:py-2 text-xs sm:text-sm font-semibold rounded-lg transition-all duration-200 bg-white text-indigo-600 shadow-sm';
                tabYoYBtn.className = 'px-3 py-1.5 sm:px-4 sm:py-2 text-xs sm:text-sm font-semibold rounded-lg transition-all duration-200 text-slate-600 hover:text-slate-900';
            } else {
                viewMain.classList.add('hidden');
                viewYoY.classList.remove('hidden');
                tabYoYBtn.className = 'px-3 py-1.5 sm:px-4 sm:py-2 text-xs sm:text-sm font-semibold rounded-lg transition-all duration-200 bg-white text-indigo-600 shadow-sm';
                tabMainBtn.className = 'px-3 py-1.5 sm:px-4 sm:py-2 text-xs sm:text-sm font-semibold rounded-lg transition-all duration-200 text-slate-600 hover:text-slate-900';
                updateYoYView();
            }
        }

        function openConfigModal() {
            const modal = document.getElementById('configModal');
            if (!modal) return;

            const inputUrlMain = document.getElementById('inputUrlMain');
            const inputUrl2025 = document.getElementById('inputUrl2025');

            if (inputUrlMain) inputUrlMain.value = localStorage.getItem('csv_url_main') || DEFAULT_CSV_URL;
            if (inputUrl2025) inputUrl2025.value = localStorage.getItem('csv_url_2025') || DEFAULT_2025_CSV_URL;

            populateModalDropdowns();
            modal.classList.remove('hidden');
        }

        function closeConfigModal() {
            const modal = document.getElementById('configModal');
            if (modal) modal.classList.add('hidden');
        }

        function populateModalDropdowns() {
            const fills = [
                { id: 'mapSales', key: 'sales' },
                { id: 'mapDate', key: 'date' },
                { id: 'mapStore', key: 'store' },
                { id: 'mapEnv', key: 'env' },
                { id: 'mapProductType', key: 'productType' },
                { id: 'mapProduct', key: 'product' },
                { id: 'mapQty', key: 'qty' }
            ];

            fills.forEach(item => {
                const select = document.getElementById(item.id);
                if (!select) return;
                select.innerHTML = '';
                columnsList.forEach(col => {
                    const opt = document.createElement('option');
                    opt.value = col;
                    opt.innerText = col;
                    if (columnMapping[item.key] === col) opt.selected = true;
                    select.appendChild(opt);
                });
            });
        }

        function saveConfigAndRefresh() {
            const inputUrlMain = document.getElementById('inputUrlMain');
            const inputUrl2025 = document.getElementById('inputUrl2025');

            if (inputUrlMain) localStorage.setItem('csv_url_main', inputUrlMain.value.trim());
            if (inputUrl2025) localStorage.setItem('csv_url_2025', inputUrl2025.value.trim());

            const mapSales = document.getElementById('mapSales');
            const mapDate = document.getElementById('mapDate');
            const mapStore = document.getElementById('mapStore');
            const mapEnv = document.getElementById('mapEnv');
            const mapProductType = document.getElementById('mapProductType');
            const mapProduct = document.getElementById('mapProduct');
            const mapQty = document.getElementById('mapQty');

            if (mapSales) columnMapping.sales = mapSales.value;
            if (mapDate) columnMapping.date = mapDate.value;
            if (mapStore) columnMapping.store = mapStore.value;
            if (mapEnv) columnMapping.env = mapEnv.value;
            if (mapProductType) columnMapping.productType = mapProductType.value;
            if (mapProduct) columnMapping.product = mapProduct.value;
            if (mapQty) columnMapping.qty = mapQty.value;

            closeConfigModal();
            loadDashboardData();
        }
